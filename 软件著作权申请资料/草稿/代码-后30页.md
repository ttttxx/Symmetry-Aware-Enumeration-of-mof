            if candidate < canonical:

    def _compute_orbits_by_gen(self, generators: List[Tuple[int, ...]]) -> List[List[int]]:
        parent = list(range(self.n))
        def find(i):
            if parent[i] == i: return i
            parent[i] = find(parent[i])
            return parent[i]
        def union(i, j):
            root_i = find(i)
            root_j = find(j)
            if root_i != root_j: parent[root_i] = root_j
        for p in generators:
            for i, target in enumerate(p):
                union(i, target)
        orbit_dict = defaultdict(list)
        for i in range(self.n):
            orbit_dict[find(i)].append(i)
        return sorted(list(orbit_dict.values()), key=lambda x: x[0])

class Node():

    def identify_node_connection_points(self):
        '''
        function:  reads a CIF file and identifies the number of connection points (nodes) that have a label containing the letter 'X'.
        :return: the number of nodes in the node file that have a label containing the letter 'X'.
        :parameter: None
        '''
        path = self.node_path
        with open(path, 'r') as cif:

            cif = filter(None, cif.split('\n'))
        num_x_nodes = 0
        for line in cif:
            line_tokens = line.split()
            if judge_vertex(line_tokens):
                if re.sub('[^a-zA-Z]', '', line_tokens[0]) == 'X':
                    num_x_nodes += 1
        return num_x_nodes

            u, v = sorted((int(i), int(j)))

                dist = structure.get_distance(i, j)
            except:
                dist = 0.0
            edges.append({
                "source": u,
                "target": v,
                "length": float(dist),
                "bond_type": "coordination"
            })
            added.add((u, v))
    return edges

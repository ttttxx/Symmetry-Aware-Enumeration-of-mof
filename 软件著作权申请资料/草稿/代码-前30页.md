def fractional_to_cartesian_matrix(a, b, c, alpha, beta, gamma):
    """将晶胞参数转换为笛卡尔变换矩阵 (3x3)。
    Returns:
        3x3矩阵 M, 满足 Cartesian = M @ Fractional。
        M 的列向量为晶格基矢 a, b, c 在笛卡尔坐标系中的表示。
    """
    pi = np.pi
    ax = a
    ay = 0.0
    az = 0.0
    bx = b * np.cos(gamma * pi / 180.0)
    by = b * np.sin(gamma * pi / 180.0)
    bz = 0.0
    cx = c * np.cos(beta * pi / 180.0)
    cy = (c * b * np.cos(alpha * pi / 180.0) - bx * cx) / by
    cz = (c ** 2.0 - cx ** 2.0 - cy ** 2.0) ** 0.5
    return np.asarray([[ax, ay, az], [bx, by, bz], [cx, cy, cz]]).T

def cycles_to_perm(cycles: List[Tuple[int, ...]], n: int) -> List[int]:
    perm = list(range(n))
    for cycle in cycles:
        l = len(cycle)
        for k in range(l):
            perm[cycle[k]] = cycle[(k + 1) % l]
    return perm
def perm_to_cycles(perm: Tuple[int, ...], n: int) -> List[List[int]]:
    visited = [False] * n
    cycles = []
    for i in range(n):
        if not visited[i]:
            curr = i
            cycle = []
            while not visited[curr]:
                visited[curr] = True
                cycle.append(curr)
                curr = perm[curr]
            cycles.append(cycle)
    return cycles
